### 🔧 修改1: loadDiscussions 函数（第288-303行）

#### 原代码：
```javascript
// 加载相关讨论
const loadDiscussions = async () => {
	try {
		const res = await getDiscussionsApi({
			messageId: messageId.value,
			limit: 10
		})

		if (checkApiResponse(res)) {
			const data = getResponseData(res)
			discussions.value = data.discussions || []
		}
	} catch (error) {
		console.error('加载讨论失败:', error)
	}

```

#### 替换为：
```javascript
// 加载相关讨论
const loadDiscussions = async () => {
	// 防止重复加载
	if (discussionsLoading.value) return

	discussionsLoading.value = true
	discussionsError.value = null

	try {
		const res = await getDiscussionsApi({
			messageId: messageId.value,
			limit: 10
		})

		if (checkApiResponse(res)) {
			const data = getResponseData(res)
			discussions.value = data.discussions || []
			discussionsError.value = null
		} else {
			throw new Error(getResponseMessage(res, '加载讨论失败'))
		}
	} catch (error) {
		console.error('加载讨论失败:', error)
		discussionsError.value = error.message || '讨论加载失败，请稍后重试'
		uni.showToast({
			title: '讨论加载失败',
			icon: 'none',
			duration: 2000
		})
	} finally {
		discussionsLoading.value = false
	}
}
```
