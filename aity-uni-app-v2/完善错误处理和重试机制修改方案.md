### 修改后代码：
```javascript
const loadDiscussions = async () => {
	// 防止重复加载
	if (discussionsLoading.value) return

	discussionsLoading.value = true
	discussionsError.value = null

	try {
		const res = await getDiscussionsApi({
			messageId: messageId.value,
			limit: 10
		})

		if (res.success) {
			discussions.value = res.data.discussions || []
			discussionsError.value = null
		} else {
			throw new Error(res.message || '加载讨论失败')
		}
	} catch (error) {
		console.error('加载讨论失败:', error)
		discussionsError.value = error.message || '讨论加载失败，请稍后重试'
		uni.showToast({
			title: '讨论加载失败',
			icon: 'none',
			duration: 2000
		})
	} finally {
		discussionsLoading.value = false
	}
}
```

### 3. 增强分享错误处理
```javascript
// 分享消息 - 添加更详细的错误处理
const handleShare = async () => {
	if (shareLoading.value) return

	shareLoading.value = true

	try {
		const shareText = `${message.value.title}\n\n${message.value.content.substring(0, 100)}...`

		await new Promise((resolve, reject) => {
			uni.setClipboardData({
				data: shareText,
				success: () => {
					uni.showModal({
						title: '分享成功',
						content: '内容已复制到剪贴板，可以粘贴分享给好友',
						showCancel: false
					})
					resolve()
				},
				fail: (err) => {
					console.error('复制失败:', err)
					uni.showToast({
						title: '复制失败，请重试',
						icon: 'none',
						duration: 2000
					})
					reject(err)
				}
			})
		})
	} catch (error) {
		console.error('分享失败:', error)
	} finally {
		setTimeout(() => {
			shareLoading.value = false
		}, 500)
	}
}
```
