### 链接检查脚本

```bash
#!/bin/bash
# check-links.sh

# 查找所有 Markdown 文件
find docs -name "*.md" -type f | while read file; do
  # 提取所有链接
  grep -oP '\[.*?\]\(\K[^)]+' "$file" | while read link; do
    # 检查链接是否存在
    if [[ ! -e "$link" ]]; then
      echo "断链: $file -> $link"
    fi
  done
done
```
